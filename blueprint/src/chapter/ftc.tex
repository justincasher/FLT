\chapter{The Fundamental Theorem of Calculus}

This blueprint presents a proof of both parts of the Fundamental Theorem of Calculus (FTC).
We work over $\R$ with the standard Lebesgue measure and the interval integral.

\section{Preliminaries}

\begin{definition}[Continuous function on a closed interval]
  \label{def:continuous_on}
  \uses{}
  Let $f : [a,b] \to \R$. We say $f$ is \emph{continuous on $[a,b]$} if $f$ is continuous
  at every point of $[a,b]$ (using the subspace topology).
\end{definition}

\begin{definition}[Interval integral]
  \label{def:interval_integral}
  \uses{}
  For a function $f : \R \to \R$ and $a, b \in \R$, the \emph{interval integral}
  $\int_a^b f(x)\,dx$ is defined as $\int_a^b f\,d\mu$ where $\mu$ is the Lebesgue measure.
  By convention, $\int_a^b f = -\int_b^a f$ when $a > b$.
\end{definition}

\begin{definition}[Has derivative at]
  \label{def:has_deriv_at}
  \uses{}
  We say $F : \R \to \R$ \emph{has derivative} $f'$ \emph{at} $x$ if
  \[
    \lim_{h \to 0} \frac{F(x+h) - F(x)}{h} = f'.
  \]
\end{definition}

\begin{definition}[Antiderivative]
  \label{def:antiderivative}
  \uses{def:has_deriv_at}
  A function $F : [a,b] \to \R$ is an \emph{antiderivative} of $f : [a,b] \to \R$ if
  $F$ is differentiable on $[a,b]$ and $F'(x) = f(x)$ for all $x \in [a,b]$.
\end{definition}

\section{Key lemmas}

\begin{lemma}[Continuous functions are integrable]
  \label{lem:continuous_integrable}
  \uses{def:continuous_on, def:interval_integral}
  If $f : [a,b] \to \R$ is continuous on the compact interval $[a,b]$, then $f$ is
  integrable on $[a,b]$.
\end{lemma}
\begin{proof}
  A continuous function on a compact set is bounded and measurable, hence integrable
  with respect to the Lebesgue measure.
\end{proof}

\begin{lemma}[Integral bounds]
  \label{lem:integral_bound}
  \uses{def:interval_integral, lem:continuous_integrable}
  If $f$ is integrable on $[a,b]$ and $|f(x)| \le M$ for all $x \in [a,b]$, then
  \[
    \left|\int_a^b f(x)\,dx\right| \le M \cdot (b - a).
  \]
\end{lemma}
\begin{proof}
  This follows from the monotonicity of the integral:
  $-M \le f(x) \le M$ implies $-M(b-a) \le \int_a^b f \le M(b-a)$.
\end{proof}

\begin{lemma}[Continuous functions satisfy the $\varepsilon$-$\delta$ property uniformly on compact sets]
  \label{lem:uniform_continuity}
  \uses{def:continuous_on}
  If $f$ is continuous on the compact interval $[a,b]$, then $f$ is uniformly continuous
  on $[a,b]$: for every $\varepsilon > 0$ there exists $\delta > 0$ such that
  $|x - y| < \delta$ implies $|f(x) - f(y)| < \varepsilon$ for all $x, y \in [a,b]$.
\end{lemma}
\begin{proof}
  This is the Heine--Cantor theorem: continuous functions on compact metric spaces are
  uniformly continuous.
\end{proof}

\section{First Fundamental Theorem of Calculus}

\begin{theorem}[First Fundamental Theorem of Calculus]
  \label{thm:ftc1}
  \uses{def:continuous_on, def:interval_integral, def:has_deriv_at, lem:continuous_integrable, lem:integral_bound, lem:uniform_continuity}
  Let $f : [a,b] \to \R$ be continuous on $[a,b]$. Define $F : [a,b] \to \R$ by
  \[
    F(x) = \int_a^x f(t)\,dt.
  \]
  Then $F$ is differentiable on $[a,b]$ and $F'(x) = f(x)$ for all $x \in [a,b]$.
\end{theorem}
\begin{proof}
  Fix $x \in [a,b]$. We must show
  \[
    \lim_{h \to 0} \frac{F(x+h) - F(x)}{h} = f(x).
  \]
  We have
  \[
    \frac{F(x+h) - F(x)}{h} = \frac{1}{h}\int_x^{x+h} f(t)\,dt.
  \]
  Since $f(x) = \frac{1}{h}\int_x^{x+h} f(x)\,dt$, we get
  \[
    \left|\frac{F(x+h) - F(x)}{h} - f(x)\right|
    = \left|\frac{1}{h}\int_x^{x+h} (f(t) - f(x))\,dt\right|.
  \]
  By Lemma~\ref{lem:uniform_continuity}, given $\varepsilon > 0$, there exists $\delta > 0$
  such that $|t - x| < \delta$ implies $|f(t) - f(x)| < \varepsilon$. For $|h| < \delta$,
  by Lemma~\ref{lem:integral_bound},
  \[
    \left|\frac{1}{h}\int_x^{x+h} (f(t) - f(x))\,dt\right| \le \varepsilon.
  \]
  Hence $F'(x) = f(x)$.
\end{proof}

\section{Second Fundamental Theorem of Calculus}

\begin{theorem}[Second Fundamental Theorem of Calculus]
  \label{thm:ftc2}
  \uses{def:continuous_on, def:interval_integral, def:antiderivative, lem:continuous_integrable}
  Let $f : [a,b] \to \R$ be continuous on $[a,b]$, and let $F$ be any antiderivative of $f$
  on $[a,b]$. Then
  \[
    \int_a^b f(x)\,dx = F(b) - F(a).
  \]
\end{theorem}
\begin{proof}
  \uses{thm:ftc1}
  By Theorem~\ref{thm:ftc1}, the function $G(x) = \int_a^x f(t)\,dt$ is an antiderivative
  of $f$. Since $F$ is also an antiderivative of $f$, we have $(F - G)' = 0$ on $[a,b]$.
  By the mean value theorem, $F - G$ is constant on $[a,b]$, say $F(x) - G(x) = C$
  for all $x \in [a,b]$. Then
  \[
    F(b) - F(a) = (G(b) + C) - (G(a) + C) = G(b) - G(a) = \int_a^b f(t)\,dt - 0
    = \int_a^b f(t)\,dt. \qedhere
  \]
\end{proof}
