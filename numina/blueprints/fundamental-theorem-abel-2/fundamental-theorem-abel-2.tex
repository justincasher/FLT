\section{The Fundamental Theorem of Finitely Generated Abelian Groups}

\subsection{Definitions}

\begin{definition}[Abelian group]
    \label{def:abelian-group}
    A \emph{group} is a set $G$ with a binary operation $\cdot : G \times G \to G$
    that is associative, has a two-sided identity $e$, and admits two-sided inverses.
    It is \emph{abelian} if $a \cdot b = b \cdot a$ for all $a, b \in G$.
    Abelian groups are written additively, with identity $0$.
\end{definition}

\begin{definition}[Subgroup generated by a set]
    \label{def:subgroup-generated}
    \uses{def:abelian-group}
    For $S \subseteq G$, the \emph{subgroup generated by $S$}, written
    $\langle S \rangle$, is the intersection of all subgroups of $G$ containing $S$.
    For abelian $G$, it is the set of all finite $\mathbb{Z}$-linear combinations
    $n_1 s_1 + \cdots + n_k s_k$.
\end{definition}

\begin{definition}[Finitely generated abelian group]
    \label{def:fg-abelian}
    \uses{def:abelian-group, def:subgroup-generated}
    An abelian group $G$ is \emph{finitely generated} if $G = \langle S \rangle$
    for some finite set $S \subseteq G$.
\end{definition}

\begin{definition}[Cyclic group]
    \label{def:cyclic-group}
    \uses{def:abelian-group, def:subgroup-generated}
    A group $G$ is \emph{cyclic} if $G = \langle a \rangle$ for some $a \in G$.
    Up to isomorphism, the cyclic groups are $\mathbb{Z}$ and $\mathbb{Z}/n\mathbb{Z}$ for $n \ge 1$.
\end{definition}

\begin{definition}[Direct sum of abelian groups]
    \label{def:direct-sum}
    \uses{def:abelian-group}
    The (external) \emph{direct sum} $A_1 \oplus \cdots \oplus A_k$ is the set of
    tuples $(a_1, \dots, a_k)$ with $a_i \in A_i$, under coordinatewise addition.
\end{definition}

\begin{definition}[Free abelian group of rank $n$]
    \label{def:free-abelian}
    \uses{def:abelian-group}
    An abelian group $F$ is \emph{free of rank $n$} if $F \cong \mathbb{Z}^n$.
    Equivalently, $F$ has a basis $\{e_1, \dots, e_n\}$: a subset such that
    every element has a unique expression $\sum c_i e_i$ with $c_i \in \mathbb{Z}$.
    The rank is well-defined, equal to the $\mathbb{Q}$-dimension of $F \otimes_{\mathbb{Z}} \mathbb{Q}$.
\end{definition}

\begin{definition}[Torsion subgroup]
    \label{def:torsion}
    \uses{def:abelian-group}
    An element $g \in G$ is \emph{torsion} if $n g = 0$ for some $n \ge 1$.
    The torsion elements form a subgroup $T(G)$, the \emph{torsion subgroup};
    $G$ is \emph{torsion-free} if $T(G) = 0$.
\end{definition}

\subsection{Auxiliary Results}

\begin{lemma}[Subgroups of $\mathbb{Z}^n$ are free]
    \label{lem:fgfree}
    \uses{def:free-abelian}
    Every subgroup $H \le \mathbb{Z}^n$ is free abelian of rank at most $n$.
\end{lemma}
\begin{proof}
    \uses{def:free-abelian}
    Proceed by induction on $n$. For $n = 1$, every subgroup of $\mathbb{Z}$ equals
    $d\mathbb{Z}$ for some $d \ge 0$ (take the smallest positive element, using the
    division algorithm); this is free of rank $0$ or $1$.

    For $n > 1$, let $\pi : \mathbb{Z}^n \to \mathbb{Z}$ be the last-coordinate projection.
    Then $\pi(H) = d\mathbb{Z}$ for some $d \ge 0$. If $d = 0$, then
    $H \subseteq \mathbb{Z}^{n-1}$ and induction applies. Otherwise pick $h \in H$
    with $\pi(h) = d$, and let $K = H \cap \ker \pi \subseteq \mathbb{Z}^{n-1}$.
    By induction, $K$ has a basis $e_1, \dots, e_m$. Every $x \in H$ satisfies
    $\pi(x) = q d$ for some $q \in \mathbb{Z}$, so $x - q h \in K$ has a unique
    expansion in $e_1, \dots, e_m$. Hence $e_1, \dots, e_m, h$ is a basis of $H$.
\end{proof}

\begin{lemma}[Finite presentation]
    \label{lem:pres}
    \uses{def:fg-abelian, def:free-abelian}
    Every finitely generated abelian group $G$ fits in a short exact sequence
    \[
        0 \to K \to \mathbb{Z}^n \to G \to 0
    \]
    with $K$ free abelian of rank $m \le n$.
\end{lemma}
\begin{proof}
    \uses{lem:fgfree, def:fg-abelian}
    If $g_1, \dots, g_n$ generate $G$, the map $\varphi : \mathbb{Z}^n \to G$
    defined by $(a_i) \mapsto \sum a_i g_i$ is surjective. Set $K = \ker \varphi$
    and apply Lemma~\ref{lem:fgfree} to conclude that $K$ is free abelian of
    rank at most $n$.
\end{proof}

\begin{definition}[Smith Normal Form]
    \label{def:snf}
    A matrix $A \in M_{n \times m}(\mathbb{Z})$ is in \emph{Smith Normal Form} if
    it is zero off the main diagonal and its diagonal entries
    $d_1, \dots, d_s$ (with the remaining diagonal positions equal to $0$)
    satisfy $d_1 \mid d_2 \mid \cdots \mid d_s$ with all $d_i > 0$.
\end{definition}

\begin{lemma}[Existence of Smith Normal Form]
    \label{lem:snf-existence}
    \uses{def:snf}
    For every $A \in M_{n \times m}(\mathbb{Z})$, there exist matrices
    $U \in GL_n(\mathbb{Z})$ and $V \in GL_m(\mathbb{Z})$ such that $U A V$ is in Smith Normal Form.
\end{lemma}
\begin{proof}
    \uses{def:snf}
    Integer elementary row and column operations (row/column swaps, sign
    change, adding an integer multiple of one row/column to another) correspond
    to left/right multiplication by elements of $GL_n(\mathbb{Z})$ or $GL_m(\mathbb{Z})$.

    If $A = 0$ we are done. Otherwise let $\delta(A)$ be the smallest positive
    absolute value among the entries of $A$. Move an entry achieving it to
    position $(1,1)$ with positive sign, so $a_{11} = \delta(A) > 0$. For each
    $j > 1$, write $a_{1j} = q a_{11} + r$ with $0 \le r < a_{11}$ and subtract
    $q$ times column $1$ from column $j$; if $r > 0$ the new entry reduces
    $\delta$, so restart. Do the same for column $1$ via row operations.
    Since $\delta$ is a positive integer that strictly decreases at each
    restart, the procedure terminates with row $1$ and column $1$ zero except
    for $a_{11} > 0$.

    If some entry $a_{ij}$ with $i, j > 1$ is not divisible by $a_{11}$, add
    row $i$ to row $1$ and repeat; $\delta$ strictly decreases. After finitely
    many iterations $a_{11}$ divides all remaining entries; recursively apply
    to the submatrix obtained by deleting row $1$ and column $1$. Since
    divisibility by $a_{11}$ is preserved, the divisibility chain
    $d_1 \mid d_2 \mid \cdots$ holds.
\end{proof}

\begin{lemma}[Uniqueness of Smith Normal Form]
    \label{lem:snf-uniqueness}
    \uses{def:snf}
    The diagonal entries $d_1, \dots, d_s$ of the Smith Normal Form of $A$
    are uniquely determined by $A$.
\end{lemma}
\begin{proof}
    \uses{def:snf}
    For $1 \le k \le \min(n, m)$, let $\Delta_k(A)$ denote the gcd of all
    $k \times k$ minors of $A$, with the convention $\Delta_0(A) = 1$ and
    $\Delta_k(A) = 0$ if every $k \times k$ minor vanishes. Each elementary
    operation expresses the new $k \times k$ minors as integer linear
    combinations of the old ones (and conversely), so $\Delta_k$ is invariant
    under left/right multiplication by elements of $GL_n(\mathbb{Z})$ or $GL_m(\mathbb{Z})$.
    For a matrix in Smith form with diagonal $d_1 \mid \cdots \mid d_s$,
    one computes $\Delta_k = d_1 d_2 \cdots d_k$ for $k \le s$ and
    $\Delta_k = 0$ for $k > s$. Hence $s$ is determined by $A$, and
    $d_k = \Delta_k(A) / \Delta_{k-1}(A)$ is determined by $A$.
\end{proof}

\begin{lemma}[Smith Normal Form]
    \label{lem:snf-combined}
    \uses{def:snf, lem:snf-existence, lem:snf-uniqueness}
    For every $A \in M_{n \times m}(\mathbb{Z})$ there exist $U \in GL_n(\mathbb{Z})$
    and $V \in GL_m(\mathbb{Z})$ such that $U A V$ is in Smith Normal Form, and
    the diagonal entries are uniquely determined by $A$.
\end{lemma}
\begin{proof}
    \uses{lem:snf-existence, lem:snf-uniqueness}
    Combine Lemma~\ref{lem:snf-existence} (existence) and
    Lemma~\ref{lem:snf-uniqueness} (uniqueness).
\end{proof}

\subsection{Existence of the Decomposition}

\begin{proposition}[Existence of invariant factor decomposition]
    \label{prop:existence}
    \uses{def:fg-abelian, def:free-abelian, def:cyclic-group, def:direct-sum}
    Let $G$ be a finitely generated abelian group. There exist integers
    $r \ge 0$, $k \ge 0$, and $n_1, \dots, n_k \ge 2$ with
    $n_1 \mid n_2 \mid \cdots \mid n_k$ such that
    \[
        G \;\cong\; \mathbb{Z}^r \oplus \mathbb{Z}/n_1\mathbb{Z} \oplus \mathbb{Z}/n_2\mathbb{Z}
        \oplus \cdots \oplus \mathbb{Z}/n_k\mathbb{Z}.
    \]
\end{proposition}
\begin{proof}
    \uses{lem:pres, lem:snf-combined}
    By Lemma~\ref{lem:pres}, $G \cong \mathbb{Z}^n / K$ with $K \le \mathbb{Z}^n$ free of
    rank $m \le n$. Choose a basis $f_1, \dots, f_m$ of $K$ and the standard
    basis $e_1, \dots, e_n$ of $\mathbb{Z}^n$. Writing $f_j = \sum_i A_{ij} e_i$
    defines a matrix $A \in M_{n \times m}(\mathbb{Z})$.

    By Lemma~\ref{lem:snf-combined} there exist $U \in GL_n(\mathbb{Z})$ and
    $V \in GL_m(\mathbb{Z})$ with $U A V$ in Smith form, with diagonal
    $d_1 \mid \cdots \mid d_s$, $s \le m$. The columns of $U^{-1}$ form a new
    basis $e_1', \dots, e_n'$ of $\mathbb{Z}^n$, and the columns of $V$ describe a
    new basis $f_1', \dots, f_m'$ of $K$; in these bases $f_j' = d_j e_j'$ for
    $j \le s$. Since $K$ is free of rank $m$ we have $s = m$, and
    \[
        K = \mathbb{Z}(d_1 e_1') \oplus \cdots \oplus \mathbb{Z}(d_m e_m')
        \subseteq \mathbb{Z} e_1' \oplus \cdots \oplus \mathbb{Z} e_n' = \mathbb{Z}^n,
    \]
    so
    \[
        G \cong \mathbb{Z}^n / K \cong \mathbb{Z}/d_1 \oplus \cdots \oplus \mathbb{Z}/d_m
        \oplus \mathbb{Z}^{n - m}.
    \]
    Discarding the trivial summands $\mathbb{Z}/1 = 0$ and renaming the remaining
    $d_i \ge 2$ as $n_1 \mid \cdots \mid n_k$, with $r = n - m$, gives the
    required decomposition.
\end{proof}

\subsection{Uniqueness of the Decomposition}

\begin{lemma}[Rank and torsion are intrinsic]
    \label{lem:rank}
    \uses{def:torsion, def:free-abelian, def:direct-sum}
    If $G \cong \mathbb{Z}^r \oplus T$ with $T$ finite, then $T = T(G)$ and
    $r = \dim_{\mathbb{Q}}(G \otimes_{\mathbb{Z}} \mathbb{Q})$. In particular, $r$ depends
    only on $G$.
\end{lemma}
\begin{proof}
    \uses{def:torsion, def:free-abelian}
    Every element of $T$ has finite order, while no nonzero element of $\mathbb{Z}^r$
    does. So $T(G) = T$. Tensoring with $\mathbb{Q}$ kills the (finite) torsion and
    gives $\mathbb{Q}^r$; the $\mathbb{Q}$-dimension is an isomorphism invariant of $G$.
\end{proof}

\begin{lemma}[$p$-primary dimensions]
    \label{lem:pdim}
    \uses{def:direct-sum, def:cyclic-group}
    Let $G \cong \mathbb{Z}/n_1 \oplus \cdots \oplus \mathbb{Z}/n_k$ be a finite abelian
    group written with $n_1 \mid \cdots \mid n_k$, and let $p$ be a prime and
    $i \ge 0$. The quotient
    \[
        U_i(G) := p^i G \,/\, p^{i+1} G
    \]
    is naturally an $\mathbb{F}_p$-vector space, and
    \[
        \dim_{\mathbb{F}_p} U_i(G) = \#\{\, j : p^{i+1} \mid n_j \,\}.
    \]
\end{lemma}
\begin{proof}
    \uses{def:cyclic-group, def:direct-sum}
    Multiplication by $p$ annihilates $U_i(G)$, giving the $\mathbb{F}_p$-structure.
    Direct sums commute with the operations $p^i(-)$ and quotients, so it
    suffices to compute the contribution of a single summand $C = \mathbb{Z}/n$.
    We have $p^i C = (p^i \mathbb{Z} + n\mathbb{Z})/n\mathbb{Z} = \gcd(p^i, n)\mathbb{Z}/n\mathbb{Z}$, of order
    $n / \gcd(p^i, n)$. Thus
    \[
        |p^i C / p^{i+1} C| = \frac{|p^i C|}{|p^{i+1} C|}
        = \frac{\gcd(p^{i+1}, n)}{\gcd(p^i, n)}.
    \]
    Writing $n = p^v m$ with $\gcd(p, m) = 1$: if $v \ge i + 1$ the ratio is
    $p$, otherwise it is $1$. Hence $\mathbb{Z}/n_j$ contributes $1$ to
    $\dim_{\mathbb{F}_p} U_i(G)$ exactly when $p^{i+1} \mid n_j$.
\end{proof}

\begin{proposition}[Uniqueness of invariant factors]
    \label{prop:uniqueness}
    \uses{def:fg-abelian, def:direct-sum, def:free-abelian, def:cyclic-group}
    In any decomposition
    $G \cong \mathbb{Z}^r \oplus \mathbb{Z}/n_1 \oplus \cdots \oplus \mathbb{Z}/n_k$ of a
    finitely generated abelian group with $n_1 \mid \cdots \mid n_k$ and
    $n_i \ge 2$, the integer $r$ and the sequence $(n_1, \dots, n_k)$ are
    uniquely determined by $G$.
\end{proposition}
\begin{proof}
    \uses{lem:rank, lem:pdim}
    By Lemma~\ref{lem:rank}, $r$ is determined by $G$, and the torsion part
    $T(G) \cong \mathbb{Z}/n_1 \oplus \cdots \oplus \mathbb{Z}/n_k$ is determined as a
    group. So we reduce to $G$ finite and prove uniqueness of the invariant
    factors $n_1 \mid \cdots \mid n_k$ with $n_i \ge 2$.

    Let $p$ be a prime and write $a_p(j) = v_p(n_j)$ for the $p$-adic
    valuation of $n_j$. Since $n_1 \mid \cdots \mid n_k$, we have
    $0 \le a_p(1) \le \cdots \le a_p(k)$. Lemma~\ref{lem:pdim} gives
    \[
        \dim_{\mathbb{F}_p} U_i(G) = \#\{\, j : a_p(j) \ge i + 1 \,\}.
    \]
    Viewing the right-hand side as a function of $i$ completely determines the
    nondecreasing sequence $(a_p(1), \dots, a_p(k))$: the count with
    $a_p(j) \ge 1$ is $\dim U_0$, the count with $a_p(j) \ge 2$ is
    $\dim U_1$, and so on. Repeating for every prime $p$ recovers each
    $n_j = \prod_p p^{a_p(j)}$. Since the dimensions $\dim_{\mathbb{F}_p} U_i(G)$
    depend only on $G$ as an abstract abelian group, the invariant factors
    are unique.
\end{proof}

\subsection{Main Theorem}

\begin{theorem}[Fundamental Theorem of Finitely Generated Abelian Groups]
    \label{thm:main}
    \uses{def:fg-abelian, def:free-abelian, def:cyclic-group, def:direct-sum,
          prop:existence, prop:uniqueness}
    Let $G$ be a finitely generated abelian group. There exist unique integers
    $r \ge 0$, $k \ge 0$, and $n_1, \dots, n_k \ge 2$ with
    $n_1 \mid n_2 \mid \cdots \mid n_k$ such that
    \[
        G \;\cong\; \mathbb{Z}^r \oplus \mathbb{Z}/n_1\mathbb{Z} \oplus \mathbb{Z}/n_2\mathbb{Z}
        \oplus \cdots \oplus \mathbb{Z}/n_k\mathbb{Z}.
    \]
    The integer $r$ is the \emph{free rank}; the $n_i$ are the
    \emph{invariant factors}.
\end{theorem}
\begin{proof}
    \uses{prop:existence, prop:uniqueness}
    Existence of the decomposition is Proposition~\ref{prop:existence}, and
    uniqueness of $r$ and the invariant factors is
    Proposition~\ref{prop:uniqueness}.
\end{proof}

\subsection{Corollary: Primary Decomposition}

\begin{corollary}[Primary decomposition of finite abelian groups]
    \label{cor:primary}
    \uses{def:abelian-group, def:cyclic-group, def:direct-sum, thm:main}
    Every finite abelian group is isomorphic to a direct sum of cyclic groups
    of prime-power order, and the multiset of prime-power orders is unique.
\end{corollary}
\begin{proof}
    \uses{thm:main}
    Apply Theorem~\ref{thm:main} to obtain $G \cong \mathbb{Z}/n_1 \oplus \cdots
    \oplus \mathbb{Z}/n_k$, and then apply the Chinese Remainder Theorem to each
    $\mathbb{Z}/n_j$ via $n_j = \prod_p p^{a_p(j)}$, regrouping. Uniqueness of the
    prime-power multiset follows from the uniqueness of the exponents
    $a_p(j)$ established in the proof of Proposition~\ref{prop:uniqueness}.
\end{proof}
