\section{Definitions}

\begin{definition}[Antiderivative]
    \label{def:antiderivative}

    A function $F : \mathbb{R} \to \mathbb{R}$ is an \emph{antiderivative} of $f$ on $(a,b)$ if $\texttt{HasDerivAt}\; F\; (f\; x)\; x$ for every $x \in (a,b)$. When $a \ge b$ the condition is vacuously true.
\end{definition}

\section{Auxiliary Lemmas}

\begin{lemma}[Continuous functions are interval integrable]
    \label{lem:continuous_interval_integrable}

    If $f : \mathbb{R} \to \mathbb{R}$ is continuous on $[\min(a,b), \max(a,b)]$, then $f$ is interval integrable on $[a,b]$ with respect to the Lebesgue measure. This is \texttt{ContinuousOn.intervalIntegrable}.
\end{lemma}
\begin{proof}
    Continuous functions on compact intervals are bounded and measurable, hence integrable.
\end{proof}

\begin{lemma}[Interval integrability restriction]
    \label{lem:interval_integrable_restriction}
    \uses{lem:continuous_interval_integrable}

    If $f$ is interval integrable on $[a,b]$ and $x \in (\min(a,b), \max(a,b))$, then $f$ is interval integrable on $[a,x]$. This follows from \texttt{IntervalIntegrable.mono\_set} since $\texttt{uIcc}\; a\; x \subseteq \texttt{uIcc}\; a\; b$.
\end{lemma}
\begin{proof}
    \uses{lem:continuous_interval_integrable}
    The interval $\texttt{uIcc}\; a\; x$ is contained in $\texttt{uIcc}\; a\; b$, so integrability restricts by \texttt{IntervalIntegrable.mono\_set}.
\end{proof}

\begin{lemma}[Strong measurability at interior points]
    \label{lem:strongly_measurable_interior}

    If $f$ is continuous on $[\min(a,b), \max(a,b)]$ and $x \in (\min(a,b), \max(a,b))$, then $f$ is strongly measurable at the neighborhood filter of $x$. This follows via \texttt{ContinuousOn.stronglyMeasurableAtFilter} since $x$ lies in the open set $\mathrm{Ioo}\;(\min\,a\,b)\;(\max\,a\,b)$.
\end{lemma}
\begin{proof}
    Apply \texttt{ContinuousOn.stronglyMeasurableAtFilter} with \texttt{isOpen\_Ioo}, since $x$ lies in the open interior.
\end{proof}

\begin{lemma}[Continuity at interior points]
    \label{lem:continuous_at_interior}

    If $f$ is continuous on $[\min(a,b), \max(a,b)]$ and $x \in (\min(a,b), \max(a,b))$, then $f$ is continuous at $x$. Since $x$ is interior, the closed interval is a neighborhood of $x$, so \texttt{ContinuousOn} upgrades to \texttt{ContinuousAt}.
\end{lemma}
\begin{proof}
    The closed interval is a neighborhood of $x$ (by \texttt{Icc\_mem\_nhds}), so continuity on the interval implies continuity at $x$.
\end{proof}

\begin{lemma}[Mean Value Theorem]
    \label{lem:mvt}

    If $G$ is continuous on $[a,b]$ with $a < b$ and $\texttt{HasDerivAt}\; G\; (G'\; x)\; x$ for all $x \in (a,b)$, then there exists $c \in (a,b)$ with $G'(c) = (G(b) - G(a))/(b - a)$. This is \texttt{exists\_hasDerivAt\_eq\_slope}.
\end{lemma}
\begin{proof}
    Direct application of \texttt{exists\_hasDerivAt\_eq\_slope}.
\end{proof}

\begin{lemma}[Zero derivative implies constant]
    \label{lem:zero_deriv_constant}
    \uses{lem:mvt}

    If $G$ is continuous on $[a,b]$ and $\texttt{HasDerivAt}\; G\; 0\; x$ for all $x \in (a,b)$, then $G(x) = G(a)$ for all $x \in [a,b]$.
\end{lemma}
\begin{proof}
    \uses{lem:mvt}
    For $x = a$ the result is immediate. For $a < x$, restrict to $[a,x]$ and apply the MVT to get $c \in (a,x)$ with $G'(c) = (G(x) - G(a))/(x-a)$. Since $G'(c) = 0$ and $x \ne a$, we get $G(x) = G(a)$.
\end{proof}

\begin{lemma}[Difference of antiderivatives has zero derivative]
    \label{lem:antideriv_diff_zero_deriv}
    \uses{def:antiderivative}

    If $F$ and $G$ are both antiderivatives of $f$ on $(a,b)$, then $\texttt{HasDerivAt}\; (F - G)\; 0\; x$ for every $x \in (a,b)$.
\end{lemma}
\begin{proof}
    \uses{def:antiderivative}
    Both $F$ and $G$ have derivative $f(x)$ at $x$, so by \texttt{HasDerivAt.sub}, $F - G$ has derivative $f(x) - f(x) = 0$.
\end{proof}

\section{Main Theorems}

\begin{theorem}[First Fundamental Theorem of Calculus]
    \label{thm:ftc1}
    \uses{lem:interval_integrable_restriction, lem:strongly_measurable_interior, lem:continuous_at_interior}

    Let $f$ be continuous on $[\min(a,b), \max(a,b)]$ and $x \in (\min(a,b), \max(a,b))$. Then $F(u) = \int_a^u f(t)\,dt$ satisfies $\texttt{HasDerivAt}\; F\; (f\; x)\; x$. The core Mathlib result is \texttt{intervalIntegral.integral\_hasDerivAt\_right}.
\end{theorem}
\begin{proof}
    \uses{lem:interval_integrable_restriction, lem:strongly_measurable_interior, lem:continuous_at_interior}
    We verify the three hypotheses of \texttt{integral\_hasDerivAt\_right}: (1) interval integrability on $[a,x]$ via Lemma~\ref{lem:interval_integrable_restriction}; (2) strong measurability at $\texttt{nhds}\;x$ via Lemma~\ref{lem:strongly_measurable_interior}; (3) continuity at $x$ via Lemma~\ref{lem:continuous_at_interior}.
\end{proof}

\begin{theorem}[Second Fundamental Theorem of Calculus]
    \label{thm:ftc2}

    Let $F$ be continuous on $[\min(a,b), \max(a,b)]$ with right derivative $f'(x)$ at every interior point $x$, and suppose $f'$ is interval integrable on $[a,b]$. Then
    \[
      \int_a^b f'(x)\,dx = F(b) - F(a).
    \]
    This is \texttt{intervalIntegral.integral\_eq\_sub\_of\_hasDeriv\_right}.
\end{theorem}
\begin{proof}
    Direct application of \texttt{integral\_eq\_sub\_of\_hasDeriv\_right}.
\end{proof}

\section{Intermediate Results}

\begin{lemma}[The integral function is an antiderivative]
    \label{lem:integral_is_antiderivative}
    \uses{def:antiderivative, thm:ftc1}

    If $f$ is continuous on $[\min(a,b), \max(a,b)]$, then $G(u) = \int_a^u f(t)\,dt$ is an antiderivative of $f$ on $(\min(a,b), \max(a,b))$.
\end{lemma}
\begin{proof}
    \uses{def:antiderivative, thm:ftc1}
    Immediate from the First FTC (Theorem~\ref{thm:ftc1}) and Definition~\ref{def:antiderivative}.
\end{proof}

\begin{lemma}[Two antiderivatives differ by a constant]
    \label{lem:antiderivatives_diff_constant}
    \uses{lem:antideriv_diff_zero_deriv, lem:zero_deriv_constant}

    If $F$ and $G$ are antiderivatives of $f$ on $(a,b)$, both continuous on $[a,b]$, then $(F-G)(x) = (F-G)(a)$ for all $x \in [a,b]$.
\end{lemma}
\begin{proof}
    \uses{lem:antideriv_diff_zero_deriv, lem:zero_deriv_constant}
    By Lemma~\ref{lem:antideriv_diff_zero_deriv}, $F-G$ has zero derivative on $(a,b)$. Since $F-G$ is continuous on $[a,b]$, Lemma~\ref{lem:zero_deriv_constant} gives the result.
\end{proof}

\begin{lemma}[Integral from a point to itself is zero]
    \label{lem:integral_base_zero}

    For any $f$ and $a$, $\int_a^a f(t)\,dt = 0$. This is \texttt{intervalIntegral.integral\_same}.
\end{lemma}
\begin{proof}
    The interval $[a,a]$ has zero length.
\end{proof}

\begin{lemma}[Integral function value at base point]
    \label{lem:integral_func_at_base}
    \uses{lem:integral_base_zero}

    For $G(u) = \int_a^u f(t)\,dt$, we have $G(a) = 0$.
\end{lemma}
\begin{proof}
    \uses{lem:integral_base_zero}
    Immediate from Lemma~\ref{lem:integral_base_zero}.
\end{proof}

\begin{lemma}[Constant difference rearrangement]
    \label{lem:constant_diff_rearrangement}

    If $F(b) - G(b) = F(a) - G(a)$ and $G(a) = 0$, then $G(b) = F(b) - F(a)$.
\end{lemma}
\begin{proof}
    Rearranging gives $G(b) - G(a) = F(b) - F(a)$, and substituting $G(a) = 0$ yields the result.
\end{proof}

\section{Corollaries}

\begin{corollary}[Evaluation of integrals via antiderivatives]
    \label{cor:ftc_eval}
    \uses{def:antiderivative, lem:integral_is_antiderivative, lem:antiderivatives_diff_constant, lem:integral_func_at_base, lem:constant_diff_rearrangement}

    Let $f$ be continuous on $[\min(a,b), \max(a,b)]$ and let $F$ be an antiderivative of $f$ on $(\min(a,b), \max(a,b))$, continuous on $[\min(a,b), \max(a,b)]$. Then
    \[
      \int_a^b f(x)\,dx = F(b) - F(a).
    \]
\end{corollary}
\begin{proof}
    \uses{lem:integral_is_antiderivative, lem:antiderivatives_diff_constant, lem:integral_func_at_base, lem:constant_diff_rearrangement}
    Let $G(u) = \int_a^u f(t)\,dt$. By Lemma~\ref{lem:integral_is_antiderivative}, $G$ is an antiderivative of $f$. By Lemma~\ref{lem:antiderivatives_diff_constant}, $F(b)-G(b) = F(a)-G(a)$. By Lemma~\ref{lem:integral_func_at_base}, $G(a)=0$. By Lemma~\ref{lem:constant_diff_rearrangement}, $G(b) = F(b)-F(a)$.
\end{proof}
