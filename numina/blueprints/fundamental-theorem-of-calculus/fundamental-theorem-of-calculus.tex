\section{Definitions}

\begin{definition}[Continuous on a set]
    \label{def:continuous_on}
    \lean{ContinuousOn}

    A function $f : \alpha \to \beta$ between topological spaces is \emph{continuous on} a set $s \subseteq \alpha$ if for every $x \in s$, $f$ is continuous within $s$ at $x$, i.e.\ the filter map of $f$ along the neighborhood filter of $x$ restricted to $s$ tends to $f(x)$. In Mathlib this is \texttt{ContinuousOn}.
\end{definition}

\begin{definition}[Has derivative at a point]
    \label{def:has_deriv_at}
    \lean{HasDerivAt}

    Let $F : \mathbb{F} \to \mathbb{F}$ where $\mathbb{F}$ is a nontrivially normed field. We say $F$ \emph{has derivative} $f'$ \emph{at} $x$ if the Fr\'echet derivative of $F$ at $x$ exists and equals scalar multiplication by $f'$, i.e., $F(x + h) = F(x) + f' \cdot h + o(h)$ as $h \to 0$. In Mathlib this is \texttt{HasDerivAt}.
\end{definition}

\begin{definition}[Interval integral]
    \label{def:interval_integral}
    \lean{intervalIntegral}

    For a function $f : \mathbb{R} \to E$ (where $E$ is a normed space) and $a, b \in \mathbb{R}$, the \emph{interval integral} $\int_a^b f(x)\,dx$ is defined as $\int_{[a,b]} f\,d\mu - \int_{[b,a]} f\,d\mu$ where $\mu$ is a given measure (typically the Lebesgue measure). This definition is antisymmetric: $\int_a^b f = -\int_b^a f$. In Mathlib this is \texttt{intervalIntegral}.
\end{definition}

\begin{definition}[Antiderivative]
    \label{def:antiderivative}
    \uses{def:has_deriv_at}
    \lean{FLT.FundamentalTheoremOfCalculus.IsAntiderivativeOn}

    A function $F : \mathbb{R} \to \mathbb{R}$ is an \emph{antiderivative} of $f$ on $(a,b)$ if for every $x \in (a,b)$, $F$ has derivative $f(x)$ at $x$ (in the sense of \texttt{HasDerivAt}). In the Lean formalization this is \texttt{IsAntiderivativeOn F f a b}, defined as $\forall x \in (a,b),\; \texttt{HasDerivAt}\; F\; (f\; x)\; x$.
\end{definition}

\section{Supporting Lemmas}

\begin{lemma}[Continuous functions are interval integrable]
    \label{lem:continuous_integrable}
    \uses{def:continuous_on, def:interval_integral}
    \lean{FLT.FundamentalTheoremOfCalculus.continuous_integrable}

    If $f : \mathbb{R} \to \mathbb{R}$ is continuous on the interval $[\min(a,b), \max(a,b)]$, then $f$ is interval integrable on $[a, b]$ with respect to the Lebesgue measure. In Mathlib this follows from \texttt{ContinuousOn.intervalIntegrable}.
\end{lemma}
\begin{proof}
    \uses{def:continuous_on, def:interval_integral}
    Direct application of \texttt{ContinuousOn.intervalIntegrable}: a continuous function on a compact interval is bounded and measurable, hence integrable with respect to the Lebesgue measure.
\end{proof}

\begin{lemma}[Integral norm bound]
    \label{lem:integral_bound}
    \uses{def:interval_integral}
    \lean{FLT.FundamentalTheoremOfCalculus.integral_bound}

    If $\|f(x)\| \le M$ for a.e.\ $x \in \operatorname{uIoc}(a,b)$, then
    \[
      \left\|\int_a^b f(x)\,dx\right\| \le M \cdot |b - a|.
    \]
    In Mathlib this is \texttt{intervalIntegral.norm\_integral\_le\_of\_norm\_le\_const\_ae}.
\end{lemma}
\begin{proof}
    \uses{def:interval_integral}
    Direct application of \texttt{intervalIntegral.norm\_integral\_le\_of\_norm\_le\_const\_ae}, which follows from the monotonicity of the Lebesgue integral and properties of the norm.
\end{proof}

\begin{lemma}[Mean Value Theorem]
    \label{lem:mean_value_theorem}
    \uses{def:has_deriv_at, def:continuous_on}
    \lean{FLT.FundamentalTheoremOfCalculus.mean_value_theorem}

    Let $f : \mathbb{R} \to \mathbb{R}$ be continuous on $[a,b]$ with $a < b$ and differentiable on $(a,b)$ with derivative $f'$. Then there exists $c \in (a,b)$ such that
    \[
      f'(c) = \frac{f(b) - f(a)}{b - a}.
    \]
    In Mathlib this is \texttt{exists\_hasDerivAt\_eq\_slope}.
\end{lemma}
\begin{proof}
    \uses{def:has_deriv_at, def:continuous_on}
    Direct application of \texttt{exists\_hasDerivAt\_eq\_slope}, which is the standard Mean Value Theorem proved via Rolle's theorem applied to an auxiliary function.
\end{proof}

\begin{lemma}[Uniform continuity on compact sets]
    \label{lem:uniform_continuity}
    \uses{def:continuous_on}
    \lean{FLT.FundamentalTheoremOfCalculus.uniform_continuity}

    If $f$ is continuous on a closed interval $[a,b]$, then $f$ is uniformly continuous on $[a,b]$. In Mathlib this is \texttt{IsCompact.uniformContinuousOn\_of\_continuous}.
\end{lemma}
\begin{proof}
    \uses{def:continuous_on}
    Direct application of \texttt{isCompact\_Icc.uniformContinuousOn\_of\_continuous}. This is the Heine--Cantor theorem: continuous functions on compact sets are uniformly continuous.
\end{proof}

\begin{lemma}[Zero derivative implies constant]
    \label{lem:zero_deriv_constant}
    \uses{def:has_deriv_at, def:continuous_on}
    \lean{FLT.FundamentalTheoremOfCalculus.zero_deriv_constant}

    Let $G : \mathbb{R} \to \mathbb{R}$ be continuous on $[a,b]$ and satisfy $\texttt{HasDerivAt}\; G\; 0\; x$ for all $x \in (a,b)$. Then $G(x) = G(a)$ for all $x \in [a,b]$.
\end{lemma}
\begin{proof}
    \uses{def:has_deriv_at, def:continuous_on}
    For any $x \in [a,b]$ with $a < x$, restrict $G$ to $[a,x]$. Apply \texttt{exists\_hasDerivAt\_eq\_slope} (the Mean Value Theorem) to obtain $c \in (a,x)$ with $G'(c) = (G(x) - G(a))/(x - a)$. Since $G'(c) = 0$ by hypothesis, we get $G(x) - G(a) = 0$, i.e., $G(x) = G(a)$.
\end{proof}

\section{Main Theorems}

\begin{theorem}[First Fundamental Theorem of Calculus]
    \label{thm:ftc1}
    \uses{def:continuous_on, def:interval_integral, def:has_deriv_at}
    \lean{FLT.FundamentalTheoremOfCalculus.ftc1}

    Let $f : \mathbb{R} \to \mathbb{R}$ be continuous on $[\min(a,b), \max(a,b)]$, and let $x \in (\min(a,b), \max(a,b))$. Then the function $F(u) = \int_a^u f(t)\,dt$ has derivative $f(x)$ at $x$. In Mathlib the core result is \texttt{intervalIntegral.integral\_hasDerivAt\_right}.
\end{theorem}
\begin{proof}
    \uses{def:continuous_on, def:interval_integral, def:has_deriv_at}
    Since $f$ is continuous on $[\min(a,b), \max(a,b)]$ and $x$ is in the interior, we establish three prerequisites for \texttt{intervalIntegral.integral\_hasDerivAt\_right}: (1) $f$ is interval integrable on $[a,x]$ via \texttt{ContinuousOn.intervalIntegrable}; (2) $f$ is strongly measurable at the neighborhood filter of $x$, using openness of the interior; (3) $f$ is continuous at $x$, since $x$ lies in the interior of the domain of continuity. The result then follows directly.
\end{proof}

\begin{theorem}[Second Fundamental Theorem of Calculus]
    \label{thm:ftc2}
    \uses{def:continuous_on, def:interval_integral, def:has_deriv_at}
    \lean{FLT.FundamentalTheoremOfCalculus.ftc2}

    Let $F : \mathbb{R} \to \mathbb{R}$ be continuous on $[\min(a,b), \max(a,b)]$ and have right derivative $f'(x)$ at every $x \in (\min(a,b), \max(a,b))$, i.e., $\texttt{HasDerivWithinAt}\; F\; (f'\; x)\; (x, +\infty)\; x$. If $f'$ is interval integrable on $[a,b]$, then
    \[
      \int_a^b f'(x)\,dx = F(b) - F(a).
    \]
    In Mathlib this is \texttt{intervalIntegral.integral\_eq\_sub\_of\_hasDeriv\_right}.
\end{theorem}
\begin{proof}
    \uses{def:continuous_on, def:interval_integral, def:has_deriv_at}
    Direct application of \texttt{intervalIntegral.integral\_eq\_sub\_of\_hasDeriv\_right}, given the continuity of $F$ on $[\min(a,b), \max(a,b)]$, the right-derivative hypothesis on the interior, and the integrability of $f'$.
\end{proof}
