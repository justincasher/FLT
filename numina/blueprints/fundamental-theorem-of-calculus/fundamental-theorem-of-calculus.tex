\section{Definitions}

\begin{definition}[Antiderivative]
    \label{def:antiderivative}

    Let $a, b \in \mathbb{R}$. A function $F : \mathbb{R} \to \mathbb{R}$ is an \emph{antiderivative} of $f : \mathbb{R} \to \mathbb{R}$ on the open interval $\mathrm{Ioo}\; a\; b = (a,b)$ if for every $x \in \mathrm{Ioo}\; a\; b$, $F$ has derivative $f(x)$ at $x$ (in the sense of \texttt{HasDerivAt}). In the Lean formalization this is \texttt{IsAntiderivativeOn F f a b}, defined as $\forall x \in \mathrm{Ioo}\; a\; b,\; \texttt{HasDerivAt}\; F\; (f\; x)\; x$. Note that $a$ and $b$ need not satisfy $a < b$; when $a \ge b$ the interval is empty and the condition is vacuously true.
\end{definition}

\section{Auxiliary Lemmas}

\begin{lemma}[Continuous functions are interval integrable]
    \label{lem:continuous_interval_integrable}

    Let $f : \mathbb{R} \to \mathbb{R}$ be continuous on $\texttt{uIcc}\; a\; b = [\min(a,b), \max(a,b)]$. Then $f$ is interval integrable on $[a,b]$ with respect to the Lebesgue measure (i.e.\ \texttt{IntervalIntegrable f volume a b}). In Mathlib this is \texttt{ContinuousOn.intervalIntegrable}.
\end{lemma}
\begin{proof}
    This is a standard Mathlib result. Continuous functions on compact intervals are bounded and measurable, hence integrable.
\end{proof}

\begin{lemma}[Strong measurability at interior points]
    \label{lem:strongly_measurable_interior}

    Let $f : \mathbb{R} \to \mathbb{R}$ be continuous on $[\min(a,b), \max(a,b)]$, and let $x \in (\min(a,b), \max(a,b))$. Then $f$ is strongly measurable at the neighborhood filter of $x$ within $\mathrm{Ioi}\; x$ (i.e.\ \texttt{StronglyMeasurableAtFilter f (nhdsWithin x (Ioi x))}). This follows because the interior of $[\min(a,b), \max(a,b)]$ is open and belongs to the Lebesgue measurable sets.
\end{lemma}
\begin{proof}
    Since $x$ lies in the open interior $(\min(a,b), \max(a,b))$, there is an open neighborhood of $x$ contained in the domain. Continuous functions on open sets are strongly measurable with respect to the Lebesgue measure, giving measurability at the neighborhood filter.
\end{proof}

\begin{lemma}[Continuity at interior points]
    \label{lem:continuous_at_interior}

    Let $f : \mathbb{R} \to \mathbb{R}$ be continuous on $[\min(a,b), \max(a,b)]$, and let $x \in (\min(a,b), \max(a,b))$. Then $f$ is continuous at $x$ within $\mathrm{Ioi}\; x$ (i.e.\ \texttt{ContinuousWithinAt f (Ioi x) x}). This follows from the fact that continuity on a set implies continuity at any interior point, restricted to any subset.
\end{lemma}
\begin{proof}
    Since $x$ is in the interior of $[\min(a,b), \max(a,b)]$, continuity on the closed interval implies continuity at $x$. Restricting to $\mathrm{Ioi}\; x$ preserves this, since $\mathrm{Ioi}\; x \subseteq [\min(a,b), \max(a,b)]$ near $x$.
\end{proof}

\begin{lemma}[Mean Value Theorem]
    \label{lem:mvt}

    Let $G : \mathbb{R} \to \mathbb{R}$ be continuous on $[a,b]$ and differentiable on $(a,b)$ with $a < b$. Then there exists $c \in (a,b)$ such that $G'(c) = (G(b) - G(a))/(b - a)$. In Mathlib this is \texttt{exists\_hasDerivAt\_eq\_slope}.
\end{lemma}
\begin{proof}
    This is a standard Mathlib result (\texttt{exists\_hasDerivAt\_eq\_slope}).
\end{proof}

\begin{lemma}[Zero derivative implies constant]
    \label{lem:zero_deriv_constant}

    Let $G : \mathbb{R} \to \mathbb{R}$ be continuous on $[a,b]$ and satisfy $\texttt{HasDerivAt}\; G\; 0\; x$ for all $x \in (a,b)$. Then $G(x) = G(a)$ for all $x \in [a,b]$.
\end{lemma}
\begin{proof}
    \uses{lem:mvt}
    For any $x \in [a,b]$ with $a < x$, restrict $G$ to $[a,x]$. By the Mean Value Theorem (\texttt{exists\_hasDerivAt\_eq\_slope}), there exists $c \in (a,x)$ with $G'(c) = (G(x) - G(a))/(x - a)$. Since $G'(c) = 0$ by hypothesis, we get $G(x) = G(a)$.
\end{proof}

\section{Main Theorems}

\begin{theorem}[First Fundamental Theorem of Calculus]
    \label{thm:ftc1}

    Let $f : \mathbb{R} \to \mathbb{R}$ be continuous on $[\min(a,b), \max(a,b)]$, and let $x \in (\min(a,b), \max(a,b))$. Then the function $F(u) = \int_a^u f(t)\,dt$ has derivative $f(x)$ at $x$. In Mathlib the core result is \texttt{intervalIntegral.integral\_hasDerivAt\_right}.
\end{theorem}
\begin{proof}
    \uses{lem:continuous_interval_integrable, lem:strongly_measurable_interior, lem:continuous_at_interior}
    We establish three prerequisites for \texttt{intervalIntegral.integral\_hasDerivAt\_right}: (1) $f$ is interval integrable on $[a,x]$ by Lemma~\ref{lem:continuous_interval_integrable}; (2) $f$ is strongly measurable at the neighborhood filter of $x$ by Lemma~\ref{lem:strongly_measurable_interior}; (3) $f$ is continuous at $x$ within $\mathrm{Ioi}\; x$ by Lemma~\ref{lem:continuous_at_interior}. The result then follows directly from \texttt{intervalIntegral.integral\_hasDerivAt\_right}.
\end{proof}

\begin{theorem}[Second Fundamental Theorem of Calculus]
    \label{thm:ftc2}

    Let $F : \mathbb{R} \to \mathbb{R}$ be continuous on $[\min(a,b), \max(a,b)]$ and have right derivative $f'(x)$ at every $x \in (\min(a,b), \max(a,b))$, i.e., $\texttt{HasDerivWithinAt}\; F\; (f'\; x)\; (x, +\infty)\; x$. If $f'$ is interval integrable on $[a,b]$, then
    \[
      \int_a^b f'(x)\,dx = F(b) - F(a).
    \]
    In Mathlib this is \texttt{intervalIntegral.integral\_eq\_sub\_of\_hasDeriv\_right}.
\end{theorem}
\begin{proof}
    Direct application of \texttt{intervalIntegral.integral\_eq\_sub\_of\_hasDeriv\_right}, given the continuity of $F$ on $[\min(a,b), \max(a,b)]$, the right-derivative hypothesis on the interior, and the integrability of $f'$.
\end{proof}

\section{Intermediate Results}

\begin{lemma}[The integral function is an antiderivative]
    \label{lem:integral_is_antiderivative}
    \uses{def:antiderivative, thm:ftc1}

    Let $f : \mathbb{R} \to \mathbb{R}$ be continuous on $[\min(a,b), \max(a,b)]$. Then the function $G(u) = \int_a^u f(t)\,dt$ is an antiderivative of $f$ on $(\min(a,b), \max(a,b))$, i.e., $\texttt{IsAntiderivativeOn}\; G\; f\; (\min(a,b))\; (\max(a,b))$.
\end{lemma}
\begin{proof}
    \uses{def:antiderivative, thm:ftc1}
    By the First FTC (Theorem~\ref{thm:ftc1}), for every $x \in (\min(a,b), \max(a,b))$, the function $G(u) = \int_a^u f(t)\,dt$ has derivative $f(x)$ at $x$. By Definition~\ref{def:antiderivative}, this means $G$ is an antiderivative of $f$ on $(\min(a,b), \max(a,b))$.
\end{proof}

\begin{lemma}[Two antiderivatives differ by a constant]
    \label{lem:antiderivatives_diff_constant}
    \uses{def:antiderivative, lem:zero_deriv_constant}

    Let $f : \mathbb{R} \to \mathbb{R}$ and let $F, G$ both be antiderivatives of $f$ on $(a,b)$ that are continuous on $[a,b]$. Then $F - G$ is constant on $[a,b]$, i.e., for all $x \in [a,b]$, $(F - G)(x) = (F - G)(a)$.
\end{lemma}
\begin{proof}
    \uses{def:antiderivative, lem:zero_deriv_constant}
    Since $F$ and $G$ are both antiderivatives of $f$ on $(a,b)$, for every $x \in (a,b)$ we have $\texttt{HasDerivAt}\; F\; (f\;x)\; x$ and $\texttt{HasDerivAt}\; G\; (f\;x)\; x$. By the linearity of derivatives, $F - G$ has derivative $f(x) - f(x) = 0$ at every $x \in (a,b)$. Moreover, $F - G$ is continuous on $[a,b]$ since both $F$ and $G$ are. By Lemma~\ref{lem:zero_deriv_constant}, $(F-G)(x) = (F-G)(a)$ for all $x \in [a,b]$.
\end{proof}

\begin{lemma}[Integral from a point to itself is zero]
    \label{lem:integral_base_zero}

    For any function $f : \mathbb{R} \to \mathbb{R}$ and any $a \in \mathbb{R}$, $\int_a^a f(t)\,dt = 0$. In Mathlib this is \texttt{intervalIntegral.integral\_same}.
\end{lemma}
\begin{proof}
    This is a standard Mathlib result (\texttt{intervalIntegral.integral\_same}). The interval $[a,a]$ has zero length, so the integral vanishes.
\end{proof}

\section{Corollaries}

\begin{corollary}[Evaluation of integrals via antiderivatives]
    \label{cor:ftc_eval}
    \uses{def:antiderivative, lem:integral_is_antiderivative, lem:antiderivatives_diff_constant, lem:integral_base_zero}

    Let $f : \mathbb{R} \to \mathbb{R}$ be continuous on $[\min(a,b), \max(a,b)]$ and let $F$ be an antiderivative of $f$ on $(\min(a,b), \max(a,b))$ that is continuous on $[\min(a,b), \max(a,b)]$. Then
    \[
      \int_a^b f(x)\,dx = F(b) - F(a).
    \]
\end{corollary}
\begin{proof}
    \uses{lem:integral_is_antiderivative, lem:antiderivatives_diff_constant, lem:integral_base_zero}
    By Lemma~\ref{lem:integral_is_antiderivative}, the function $G(u) = \int_a^u f(t)\,dt$ is an antiderivative of $f$ on the interior. By Lemma~\ref{lem:antiderivatives_diff_constant}, $F - G$ is constant on $[\min(a,b), \max(a,b)]$, so $F(b) - G(b) = F(a) - G(a)$, which gives $G(b) - G(a) = F(b) - F(a)$. By Lemma~\ref{lem:integral_base_zero}, $G(a) = \int_a^a f = 0$, so $\int_a^b f(x)\,dx = G(b) = G(b) - G(a) = F(b) - F(a)$.
\end{proof}
